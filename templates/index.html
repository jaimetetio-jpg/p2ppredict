// Función global para mostrar alertas flotantes (Toasts) function mostrarToast(mensaje, tipo = 'info') { // Evita duplicados si ya existe un contenedor o toast activo let toastContainer = document.getElementById('toast-container'); if (!toastContainer) { toastContainer = document.createElement('div'); toastContainer.id = 'toast-container'; toastContainer.style.cssText = ` position: fixed; bottom: 20px; right: 20px; z-index: 9999; display: flex; flex-direction: column; gap: 10px; `; document.body.appendChild(toastContainer); } const toast = document.createElement('div'); // Colores según el tipo de mensaje let backgroundColor = '#333'; if (tipo === 'success') backgroundColor = '#28a745'; if (tipo === 'error') backgroundColor = '#dc3545'; if (tipo === 'warning') backgroundColor = '#ffc107'; toast.style.cssText = ` background-color: ${backgroundColor}; color: #fff; padding: 12px 20px; border-radius: 6px; box-shadow: 0 4px 6px rgba(0,0,0,0.2); font-family: sans-serif; font-size: 14px; transition: opacity 0.3s ease; opacity: 0; `; toast.textContent = mensaje; toastContainer.appendChild(toast); // Animación de aparición setTimeout(() => { toast.style.opacity = '1'; }, 10); // Desaparecer y eliminar después de 4 segundos setTimeout(() => { toast.style.opacity = '0'; setTimeout(() => { toast.remove(); }, 300); }, 4000); }
